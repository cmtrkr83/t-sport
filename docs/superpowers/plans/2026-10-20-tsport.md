# TSport — X Hashtag Link Bulucu Uygulama Planı

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Android telefon ve TV'de çalışan, kullanıcının X oturumuyla `#etiket "broadcasts" filter:links` araması yapıp en fazla 20 dış linki listeleyen tek APK'lık uygulama üretmek.

**Architecture:** İki Activity (`MainActivity`, `LoginActivity`) ve görünmez bir WebView. Arama sonuçları önce DOM'dan JS ile kazınır; yetersizse sayfanın kendi `SearchTimeline` isteği yakalanıp sayfa bağlamında tekrar oynatılır. Linkler `LinkExtractor`'da filtrelenir, t.co çözülür, tekilleştirilir ve 20 ile sınırlanır.

**Tech Stack:** Kotlin 1.9.24, AGP 8.6.1, Gradle 8.7, JDK 17, Android Views + ViewBinding, RecyclerView, Coroutines, OkHttp 4.12.0, JUnit4, MockWebServer.

**Spec:** `docs/superpowers/specs/2026-10-20-tsport-design.md`

```kotlin
package com.tsport.links.search

import java.net.URLEncoder

object SearchQueryBuilder {

    fun buildSearchUrl(rawHashtag: String): String? {
        val tag = rawHashtag.trim().removePrefix("#").replace(Regex("\\s+"), "")
        if (tag.isEmpty()) return null
        val query = URLEncoder.encode("#$tag \"broadcasts\" filter:links", "UTF-8")
            .replace("+", "%20")
        return "https://x.com/search?q=$query&f=live"
    }
}
```

```kotlin
package com.tsport.links.util

object CookieUtils {

    fun hasAuthToken(cookieHeader: String?): Boolean = valueOf(cookieHeader, "auth_token") != null

    fun csrfToken(cookieHeader: String?): String? = valueOf(cookieHeader, "ct0")

    private fun valueOf(cookieHeader: String?, name: String): String? {
        if (cookieHeader.isNullOrBlank()) return null
        return cookieHeader.split(';')
            .map { it.trim() }
            .firstOrNull { it.startsWith("$name=") }
            ?.substringAfter('=')
            ?.takeIf { it.isNotEmpty() }
    }
}
```

```kotlin
package com.tsport.links.auth

import android.annotation.SuppressLint
import android.graphics.Bitmap
import android.os.Bundle
import android.webkit.CookieManager
import android.webkit.WebView
import android.webkit.WebViewClient
import android.widget.Toast
import androidx.appcompat.app.AppCompatActivity
import com.tsport.links.R
import com.tsport.links.databinding.ActivityLoginBinding

class LoginActivity : AppCompatActivity() {

    private lateinit var binding: ActivityLoginBinding
    private val authRepository = AuthRepository()

    @SuppressLint("SetJavaScriptEnabled")
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        binding = ActivityLoginBinding.inflate(layoutInflater)
        setContentView(binding.root)

        CookieManager.getInstance().setAcceptCookie(true)
        CookieManager.getInstance().setAcceptThirdPartyCookies(binding.webView, true)
        binding.webView.settings.javaScriptEnabled = true
        binding.webView.settings.domStorageEnabled = true
        binding.webView.webViewClient = object : WebViewClient() {
            override fun onPageStarted(view: WebView?, url: String?, favicon: Bitmap?) {
                binding.progress.visibility = android.view.View.VISIBLE
            }

            override fun onPageFinished(view: WebView?, url: String?) {
                binding.progress.visibility = android.view.View.GONE
                if (authRepository.isLoggedIn()) {
                    Toast.makeText(this@LoginActivity, R.string.login_success, Toast.LENGTH_SHORT).show()
                    finish()
                }
            }
        }
        binding.webView.loadUrl("https://x.com/login")
    }

    override fun onBackPressed() {
        if (binding.webView.canGoBack()) binding.webView.goBack() else finishAffinity()
    }
}
```

```kotlin
package com.tsport.links.links

import java.net.URI

class LinkExtractor(
    private val resolver: UrlResolver,
    private val maxLinks: Int = 20
) {

    suspend fun extract(candidates: List<String>): List<LinkItem> {
        val results = LinkedHashMap<String, LinkItem>()
        for (candidate in candidates) {
            if (results.size >= maxLinks) break
            val normalized = normalize(candidate) ?: continue
            val resolved = if (isTco(normalized)) resolver.resolve(normalized) else null
            val finalUrl = normalize(resolved ?: normalized) ?: normalized
            if (isExcludedHost(finalUrl)) continue
            if (results.containsKey(finalUrl)) continue
            val domain = hostOf(finalUrl) ?: continue
            results[finalUrl] = LinkItem(url = finalUrl, domain = domain)
        }
        return results.values.toList()
    }

    private fun normalize(raw: String): String? {
        var value = raw.trim()
        if (value.isEmpty()) return null
        value = value.trimEnd('.', ',', ';', ':', '!', '?', ')', ']', '}', '\'', '"')
        if (!value.startsWith("http://") && !value.startsWith("https://")) return null
        val withoutFragment = value.substringBefore('#')
        return try {
            val uri = URI(withoutFragment)
            if (uri.host.isNullOrBlank()) return null
            withoutFragment
        } catch (_: Exception) {
            null
        }
    }

    private fun isTco(url: String): Boolean = hostOf(url) == "t.co"

    private fun isExcludedHost(url: String): Boolean {
        val host = hostOf(url) ?: return true
        return host == "x.com" ||
            host.endsWith(".x.com") ||
            host == "twitter.com" ||
            host.endsWith(".twitter.com") ||
            host == "twimg.com" ||
            host.endsWith(".twimg.com")
    }

    private fun hostOf(url: String): String? = try {
        URI(url).host?.lowercase()?.removePrefix("www.")
    } catch (_: Exception) {
        null
    }
}
```

```kotlin
import android.graphics.Bitmap
import android.webkit.WebResourceError
import android.webkit.WebResourceRequest
import android.webkit.WebView
import android.webkit.WebViewClient
import kotlinx.coroutines.CompletableDeferred
import kotlinx.coroutines.Dispatchers

import kotlinx.coroutines.suspendCancellableCoroutine
import kotlinx.coroutines.withContext
import kotlinx.coroutines.withTimeout
import org.json.JSONArray
import kotlin.coroutines.resume
import kotlin.coroutines.resumeWithException

class DomLinkScraper(
    private val webView: WebView,
    private val bridge: ScrapeBridge,
    private val script: String
) {

    suspend fun scrape(
        searchUrl: String,
        maxCandidates: Int = 60,
        timeoutMs: Long = 25_000
    ): List<String> = withTimeout(timeoutMs) {
        withContext(Dispatchers.Main) {
            loadAndWait(searchUrl)
            delay(2_000)
            val deferred = CompletableDeferred<String>()
            bridge.onDomResult = { json -> deferred.complete(json) }
            try {
                val js = script.replace("__MAX__", maxCandidates.toString())
                webView.evaluateJavascript(js, null)
                parseCandidates(deferred.await())
            } finally {
                bridge.onDomResult = null
            }
        }
    }

    private suspend fun loadAndWait(url: String) = suspendCancellableCoroutine<Unit> { continuation ->
        webView.webViewClient = object : WebViewClient() {
            override fun onPageStarted(view: WebView?, url: String?, favicon: Bitmap?) = Unit

            override fun onPageFinished(view: WebView?, url: String?) {
                if (continuation.isActive) continuation.resume(Unit)
            }

            override fun onReceivedError(
                view: WebView?,
                request: WebResourceRequest?,
                error: WebResourceError?
            ) {
                if (request?.isForMainFrame == true && continuation.isActive) {
                    continuation.resumeWithException(PageLoadException())
                }
            }
        }
        webView.loadUrl(url)
    }

    private fun parseCandidates(json: String): List<String> = try {
        val array = JSONArray(json)
        (0 until array.length()).mapNotNull { index ->
            array.optString(index).takeIf { it.isNotBlank() }
        }
    } catch (_: Exception) {
        emptyList()
    }
}
```

```kotlin
package com.tsport.links.search

import org.json.JSONArray
import org.json.JSONObject
import org.json.JSONTokener

object TweetJsonLinkParser {

    private const val MAX_DEPTH = 60
    private const val MAX_URLS = 2_000

    fun extractUrls(json: String?): List<String> {
        if (json.isNullOrBlank()) return emptyList()
        val urls = LinkedHashSet<String>()
        return try {
            walk(JSONTokener(json).nextValue(), urls, 0)
            urls.toList()
        } catch (_: Exception) {
            emptyList()
        }
    }

    private fun walk(value: Any?, urls: MutableSet<String>, depth: Int) {
        if (value == null || depth > MAX_DEPTH || urls.size >= MAX_URLS) return
        when (value) {
            is JSONObject -> {
                val keys = value.keys()
                while (keys.hasNext()) {
                    val key = keys.next()
                    walk(value.opt(key), urls, depth + 1)
                }
            }
            is JSONArray -> {
                for (index in 0 until value.length()) {
                    walk(value.opt(index), urls, depth + 1)
                }
            }
            is String -> {
                if (value.startsWith("http://") || value.startsWith("https://")) {
                    urls.add(value)
                }
            }
        }
    }
}
```

`SearchViewModel.kt`:
```kotlin
package com.tsport.links.search

import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import com.tsport.links.links.LinkExtractor
import java.io.IOException
import kotlinx.coroutines.CancellationException
import kotlinx.coroutines.Job
import kotlinx.coroutines.TimeoutCancellationException
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.launch
import kotlinx.coroutines.withTimeout

class SearchViewModel(
    private val extractor: LinkExtractor
) : ViewModel() {

    private val _state = MutableStateFlow<SearchState>(SearchState.Idle)
    val state: StateFlow<SearchState> = _state.asStateFlow()

    private var job: Job? = null

    fun search(
        rawHashtag: String,
        candidateSource: suspend (searchUrl: String) -> List<String>
    ) {
        val searchUrl = SearchQueryBuilder.buildSearchUrl(rawHashtag)
        if (searchUrl == null) {
            _state.value = SearchState.Failure("Geçerli bir hashtag girin.")
            return
        }
        job?.cancel()
        job = viewModelScope.launch {
            _state.value = SearchState.Searching
            _state.value = try {
                val candidates = withTimeout(SEARCH_TIMEOUT_MS) { candidateSource(searchUrl) }
                val links = extractor.extract(candidates)
                if (links.isEmpty()) SearchState.Empty else SearchState.Success(links)
            } catch (error: TimeoutCancellationException) {
                SearchState.Failure("Arama zaman aşımına uğradı.")
            } catch (error: SessionExpiredException) {
                SearchState.Failure("Oturum sona erdi, tekrar giriş yapın.")
            } catch (error: PageLoadException) {
                SearchState.Failure("X sayfası yüklenemedi.")
            } catch (error: CancellationException) {
                throw error
            } catch (error: IOException) {
                SearchState.Failure("İnternet bağlantısı yok.")
            } catch (error: Exception) {
                SearchState.Failure("Arama başarısız oldu.")
            }
        }
    }

    fun fail(message: String) {
        _state.value = SearchState.Failure(message)
    }

    override fun onCleared() {
        job?.cancel()
        super.onCleared()
    }

    companion object {
        private const val SEARCH_TIMEOUT_MS = 25_000L
    }
}
```

```kotlin
package com.tsport.links.ui

import android.view.LayoutInflater
import android.view.ViewGroup
import androidx.recyclerview.widget.RecyclerView
import com.tsport.links.databinding.ItemLinkBinding
import com.tsport.links.links.LinkItem

class LinkAdapter(
    private val onClick: (LinkItem) -> Unit
) : RecyclerView.Adapter<LinkAdapter.LinkViewHolder>() {

    private val items = mutableListOf<LinkItem>()

    fun submit(newItems: List<LinkItem>) {
        items.clear()
        items.addAll(newItems)
        notifyDataSetChanged()
    }

    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): LinkViewHolder {
        val binding = ItemLinkBinding.inflate(LayoutInflater.from(parent.context), parent, false)
        return LinkViewHolder(binding, onClick)
    }

    override fun onBindViewHolder(holder: LinkViewHolder, position: Int) {
        holder.bind(position + 1, items[position])
    }

    override fun getItemCount(): Int = items.size

    class LinkViewHolder(
        private val binding: ItemLinkBinding,
        private val onClick: (LinkItem) -> Unit
    ) : RecyclerView.ViewHolder(binding.root) {

        fun bind(number: Int, item: LinkItem) {
            binding.textNumber.text = number.toString()
            binding.textDomain.text = item.domain
            binding.textUrl.text = item.url
            binding.root.setOnClickListener { onClick(item) }
        }
    }
}
```

```kotlin
package com.tsport.links.ui

import android.annotation.SuppressLint
import android.content.ActivityNotFoundException
import android.content.Intent
import android.content.pm.PackageManager
import android.net.ConnectivityManager
import android.net.NetworkCapabilities
import android.net.Uri
import android.os.Bundle
import android.view.View
import android.view.inputmethod.InputMethodManager
import android.webkit.CookieManager
import android.webkit.WebView
import android.widget.Toast
import androidx.activity.viewModels
import androidx.appcompat.app.AppCompatActivity
import androidx.core.content.getSystemService
import androidx.lifecycle.Lifecycle
import androidx.lifecycle.lifecycleScope
import androidx.lifecycle.repeatOnLifecycle
import androidx.recyclerview.widget.LinearLayoutManager
import com.google.android.material.dialog.MaterialAlertDialogBuilder
import com.tsport.links.R
import com.tsport.links.auth.AuthRepository
import com.tsport.links.auth.LoginActivity
import com.tsport.links.databinding.ActivityMainBinding
import com.tsport.links.links.LinkExtractor
import com.tsport.links.links.LinkItem
import com.tsport.links.links.TcoResolver
import com.tsport.links.search.ApiFallbackScraper
import com.tsport.links.search.DomLinkScraper
import com.tsport.links.search.ScrapeBridge
import com.tsport.links.search.SearchState
import com.tsport.links.search.SearchViewModel
import com.tsport.links.search.SessionExpiredException
import kotlinx.coroutines.launch

class MainActivity : AppCompatActivity() {

    private lateinit var binding: ActivityMainBinding
    private val authRepository = AuthRepository()
    private val viewModel: SearchViewModel by viewModels {
        SearchViewModelFactory(LinkExtractor(TcoResolver()))
    }
    private lateinit var bridge: ScrapeBridge
    private lateinit var domScraper: DomLinkScraper
    private lateinit var fallbackScraper: ApiFallbackScraper
    private lateinit var adapter: LinkAdapter
    private var loginShown = false

    @SuppressLint("SetJavaScriptEnabled")
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        binding = ActivityMainBinding.inflate(layoutInflater)
        setContentView(binding.root)

        adapter = LinkAdapter(::openLink)
        binding.linkList.layoutManager = LinearLayoutManager(this)
        binding.linkList.adapter = adapter

        bridge = ScrapeBridge()
        domScraper = DomLinkScraper(binding.scrapeWebView, bridge, readAsset("scrape_dom.js"))
        fallbackScraper = ApiFallbackScraper(binding.scrapeWebView, bridge, readAsset("fallback_api.js"))
        configureWebView(binding.scrapeWebView, bridge)

        binding.hashtagInput.setText(prefs().getString(KEY_LAST_HASHTAG, ""))
        binding.searchButton.setOnClickListener { startSearch() }
        binding.refreshButton.setOnClickListener { startSearch() }
        binding.logoutButton.setOnClickListener { confirmLogout() }

        lifecycleScope.launch {
            repeatOnLifecycle(Lifecycle.State.STARTED) {
                viewModel.state.collect(::render)
            }
        }
    }

    override fun onResume() {
        super.onResume()
        if (!authRepository.isLoggedIn()) {
            openLogin()
        } else {
            loginShown = false
        }
    }

    @SuppressLint("SetJavaScriptEnabled")
    private fun configureWebView(webView: WebView, bridge: ScrapeBridge) {
        CookieManager.getInstance().setAcceptCookie(true)
        CookieManager.getInstance().setAcceptThirdPartyCookies(webView, true)
        webView.settings.javaScriptEnabled = true
        webView.settings.domStorageEnabled = true
        webView.settings.userAgentString = TcoResolver.USER_AGENT
        webView.addJavascriptInterface(bridge, "TSportBridge")
        webView.isFocusable = false
        webView.isClickable = false
    }

    private fun startSearch() {
        if (!isNetworkAvailable()) {
            viewModel.fail(getString(R.string.status_no_internet))
            return
        }
        if (!authRepository.isLoggedIn()) {
            viewModel.fail(getString(R.string.session_expired))
            return
        }
        val raw = binding.hashtagInput.text.toString()
        prefs().edit().putString(KEY_LAST_HASHTAG, raw).apply()
        getSystemService<InputMethodManager>()?.hideSoftInputFromWindow(binding.hashtagInput.windowToken, 0)
        viewModel.search(raw) { searchUrl ->
            val domCandidates = domScraper.scrape(searchUrl)
            if (!authRepository.isLoggedIn()) {
                throw SessionExpiredException()
            }
            if (domCandidates.size >= MAX_LINKS) {
                domCandidates
            } else {
                domCandidates + fallbackScraper.scrape(authRepository.csrfToken().orEmpty())
            }
        }
    }

    private fun render(state: SearchState) {
        val searching = state is SearchState.Searching
        binding.progress.visibility = if (searching) View.VISIBLE else View.GONE
        binding.searchButton.isEnabled = !searching
        binding.refreshButton.isEnabled = !searching
        binding.logoutButton.isEnabled = !searching
        binding.hashtagInput.isEnabled = !searching

        when (state) {
            SearchState.Idle -> {
                binding.statusText.setText(R.string.status_idle)
                adapter.submit(emptyList())
            }
            SearchState.Searching -> {
                binding.statusText.setText(R.string.status_searching)
                adapter.submit(emptyList())
            }
            is SearchState.Success -> {
                binding.statusText.text = getString(R.string.status_found, state.links.size)
                adapter.submit(state.links)
                if (isTelevision()) binding.linkList.requestFocus()
            }
            SearchState.Empty -> {
                binding.statusText.setText(R.string.status_empty)
                adapter.submit(emptyList())
            }
            is SearchState.Failure -> {
                binding.statusText.text = state.message
                adapter.submit(emptyList())
                if (state.message == getString(R.string.session_expired)) {
                    openLogin()
                }
            }
        }
    }

    private fun openLink(item: LinkItem) {
        try {
            startActivity(Intent(Intent.ACTION_VIEW, Uri.parse(item.url)))
        } catch (_: ActivityNotFoundException) {
            Toast.makeText(this, R.string.status_no_browser, Toast.LENGTH_SHORT).show()
        }
    }

    private fun confirmLogout() {
        MaterialAlertDialogBuilder(this)
            .setTitle(R.string.logout_title)
            .setMessage(R.string.logout_message)
            .setNegativeButton(R.string.cancel, null)
            .setPositiveButton(R.string.logout) { _, _ ->
                authRepository.logout()
                binding.hashtagInput.setText("")
                adapter.submit(emptyList())
                openLogin()
            }
            .show()
    }

    private fun openLogin() {
        if (loginShown) return
        loginShown = true
        startActivity(Intent(this, LoginActivity::class.java))
    }

    private fun isNetworkAvailable(): Boolean {
        val manager = getSystemService<ConnectivityManager>() ?: return true
        val network = manager.activeNetwork ?: return false
        val capabilities = manager.getNetworkCapabilities(network) ?: return false
        return capabilities.hasCapability(NetworkCapabilities.NET_CAPABILITY_INTERNET)
    }

    private fun isTelevision(): Boolean =
        packageManager.hasSystemFeature(PackageManager.FEATURE_LEANBACK)

    private fun readAsset(name: String): String =
        assets.open(name).bufferedReader().use { it.readText() }

    private fun prefs() = getSharedPreferences("tsport", MODE_PRIVATE)

    companion object {
        private const val KEY_LAST_HASHTAG = "last_hashtag"
        private const val MAX_LINKS = 20
    }
}
```
